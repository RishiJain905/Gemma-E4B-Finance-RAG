# 1.2.5 — Integration Test & Seed Data

## Objective

Create an end-to-end integration test that verifies the entire Phase 1.2 storage layer works together, then seed the database with a small set of real financial data so Phase 1.3 (Yahoo Finance ingestion) has a baseline to build on.

## Why This Matters

This is the **Phase 1.2 shakedown** — the moment we prove that:

1. SQLite schema is correct (facts store, filings are tracked, cache works)
2. ChromaDB connects to llama-server and produces meaningful embeddings
3. The unified `Store` class correctly routes operations to the right backend
4. Real financial data flows end-to-end from a fetch → parse → store → retrieve cycle

This test also seeds a handful of tickers with fundamentals manually, so you can start querying immediately when the middleware is built in Phase 1.5 — without waiting for the full ingestion pipeline to run.

def run_tests(store: Store):
    """Run integration tests against the storage layer."""
    passed = 0
    failed = 0

    def check(name, condition):
        nonlocal passed, failed
        if condition:
            print(f"  ✅ {name}")
            passed += 1
        else:
            print(f"  ❌ {name}")
            failed += 1

    print("\n── Integration Tests ──\n")

    # 1. Heartbeat
    health = store.heartbeat()
    check("SQLite health", health.get("sqlite") == True)
    check("ChromaDB health", health.get("chroma") == True)

    # 2. Save and retrieve a fundamental
    store.save_fundamental("TEST", "unit_test", 42.0, period="2026-T1")
    fact = store.get_fundamental("TEST", "unit_test")
    check("Save & retrieve fundamental", fact and fact["value"] == 42.0)
    check("Fact has correct ticker", fact and fact["ticker"] == "TEST")
    check("Fact has correct period", fact and fact["period"] == "2026-T1")

    # 3. Update a fundamental
    store.save_fundamental("TEST", "unit_test", 99.0, period="2026-T1")
    fact = store.get_fundamental("TEST", "unit_test")
    check("Update fundamental", fact and fact["value"] == 99.0)

    # 4. Get latest fundamental (no period specified)
    store.save_fundamental("TEST", "unit_test_2", 100.0, period="2026-T1")
    latest = store.get_fundamental("TEST", "unit_test_2")
    check("Get latest fundamental", latest and latest["value"] == 100.0)

    # 5. Search facts
    facts = store.sqlite.search_facts(ticker="TEST")
    check("Search facts by ticker", len(facts) >= 2)

    # 6. Save a document to ChromaDB
    doc_id = store.save_document(
        "test/integration-doc",
        "This is a test document about financial markets and investment strategies.",
        ticker="TEST",
        source="test",
        date="2026-01-01",
    )
    check("Save document to ChromaDB", doc_id == "test/integration-doc")
    check("Document count > 0", store.chroma.count() > 0)

    # 7. Search documents
    results = store.search("financial markets investment")
    check("Semantic search returns results", len(results["documents"]) > 0)
    check("Search includes ticker detection", results.get("ticker") is not None)

    # 8. Ticker detection
    detected = store._detect_ticker("NVDA")
    check("Ticker detection: NVDA", detected == "NVDA")
    detected = store._detect_ticker("What is Apple's revenue?")
    check("Ticker detection: AAPL", detected == "AAPL")
    detected = store._detect_ticker("Tell me about the Fed rate decision")
    check("Ticker detection: None for macro query", detected is None)

    # 9. Filing pipeline
    filed = store.register_filing(
        "NVDA", "10-Q", "2026-05-15", "2026-Q1",
        "test-accession-001", "https://sec.gov/..."
    )
    check("Register filing returns True for new", filed == True)

    duplicate = store.register_filing(
        "NVDA", "10-Q", "2026-05-15", "2026-Q1",
        "test-accession-001", "https://sec.gov/..."
    )
    check("Register filing returns False for duplicate", duplicate == False)

    # 10. Process a filing through the full pipeline
    store.process_filing(
        {
            "ticker": "TEST",
            "source_type": "sec",
            "filing_type": "10-K",
            "period": "2025-FY",
            "filing_date": "2025-12-31",
            "accession": "test-accession-002",
            "source_url": "https://sec.gov/test",
        },
        extracted_text="Annual report showing strong revenue growth across all segments.",
        extracted_facts=[
            {"metric": "annual_revenue", "value": 100.0, "unit": "usd", "period": "2025-FY"},
            {"metric": "annual_profit", "value": 25.0, "unit": "usd", "period": "2025-FY"},
        ],
    )
    check("Process filing stores document", store.chroma.get_document("sec/TEST/10-K-2025-FY") is not None)
    fact = store.get_fundamental("TEST", "annual_revenue")
    check("Process filing extracts facts", fact is not None and fact["value"] == 100.0)

    # 11. Cache management
    store.mark_cache_fresh("NVDA", "yfinance", 24)
    status = store.sqlite.get_cache_status("NVDA", "yfinance")
    check("Cache status exists", status is not None)
    check("Cache status is fresh", status["status"] == "fresh")

    # 12. Search with real data
    real_results = store.search("NVIDIA data center performance")
    check("Real search returns documents", len(real_results["documents"]) > 0)
    check("Real search detects ticker", real_results["ticker"] is not None)

    # Cleanup test data
    store.chroma.delete_document("test/integration-doc")
    store.chroma.delete_document("sec/TEST/10-K-2025-FY")
    with store.sqlite._connect() as conn:
        conn.execute("DELETE FROM fundamentals WHERE ticker='TEST'")
        conn.execute("DELETE FROM filings WHERE accession IN ('test-accession-001', 'test-accession-002')")
        conn.commit()

    print(f"\n── Results: {passed} passed, {failed} failed ──\n")
    return failed == 0


def seed_data(store: Store):
    """Load seed fundamentals and documents into the database."""
    print("\n── Seeding Data ──\n")

    # Seed fundamentals
    count = 0
    for row in SEED_FUNDAMENTALS:
        ticker, metric, value, unit, period, period_type, source_type = row
        store.save_fundamental(
            ticker=ticker,
            metric=metric,
            value=value,
            unit=unit,
            period=period,
            period_type=period_type,
            source_type=source_type,
        )
        count += 1
    print(f"  ✅ Seeded {count} fundamentals")

    # Seed documents
    doc_count = 0
    for doc in SEED_DOCUMENTS:
        store.save_document(
            document_id=doc["id"],
            text=doc["text"],
            ticker=doc["ticker"],
            source=doc["source"],
            date=doc["date"],
        )
        doc_count += 1
    print(f"  ✅ Seeded {doc_count} documents")

    # Mark cache as fresh for seeded tickers
    for ticker in ["NVDA", "AMD", "META", "CRWD", "AAPL", "MSFT"]:
        store.mark_cache_fresh(ticker, "seed", 168)
    print(f"  ✅ Cache marked fresh for {6} tickers")

    # Verify
    count_after = len(store.get_fundamentals_batch("NVDA", ["revenue_q1_2026"]))
    print(f"  ℹ️   NVDA fundamentals in DB: {count_after} (should be 1+ after seeded)")
    chroma_count = store.chroma.count()
    print(f"  ℹ️   Documents in ChromaDB: {chroma_count}")
    print(f"\n── Seeding Complete ──\n")


def main():
    import argparse
    parser = argparse.ArgumentParser(description="Seed test data for Phase 1.2")
    parser.add_argument("--seed-only", action="store_true",
                        help="Skip tests and only seed data")
    args = parser.parse_args()

    store = Store()

    if args.seed_only:
        seed_data(store)
        return

    # Run tests first
    success = run_tests(store)

    if success:
        print("✅ All tests passed! Proceeding to seed data...")
        seed_data(store)
    else:
        print("❌ Tests failed. Fix issues before seeding data.")
        sys.exit(1)


if __name__ == "__main__":
    main()

## What Could Go Wrong

| Problem | Likely Cause | Fix |
|---------|-------------|------|
| `ImportError: No module named 'src'` | Running from wrong directory | `cd` to project root first |
| `ConnectionError` on ChromaDB | llama-server not running | Start it: `.\scripts\serve_model.ps1 start` |
| Tests pass but seed fails | Disk space in `data/chroma/` | Check available disk space |
| Seeded data returns search with all 1.0 distances | Embedding quality issue | Verify `--pooling mean` is active and embeddings are meaningful |
| `sqlite3.OperationalError` | Database locked by another process | Close other connections. Use WAL mode. |
| ChromaDB collection already exists with wrong dimension | Collection created with old dimension | Delete `data/chroma/` directory and re-run |
| Test 15 fails (register filing duplicate) | Previous test run left data | The cleanup section deletes test data. If re-running, it should be clean. |
