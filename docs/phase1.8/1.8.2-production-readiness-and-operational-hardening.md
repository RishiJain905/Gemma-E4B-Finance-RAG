# 1.8.2 — Production Readiness & Operational Hardening

## Objective

Harden the RAG system for reliable, unattended operation — add comprehensive error handling, structured logging, startup scripts, health monitoring, and graceful degradation paths so the system can run 24/7 without manual intervention.

## Why This Matters

A research system that crashes when an API is unreachable, or silently serves stale data, is worse than no system at all — it breeds false confidence. Phase 1.8.2 ensures the system:

- **Starts reliably** — one command brings up the full stack
- **Runs unattended** — cron jobs don't silently fail
- **Degrades gracefully** — a broken data source doesn't crash the query path
- **Reports its health** — you can check at a glance whether everything is working
- **Recovers automatically** — transient failures retry, persistent failures go to the dead-letter queue

```bash
#!/bin/bash
# start_stack.sh — Bring up the full Gemma-E4B-Finance-RAG stack
set -e

SCRIPT_DIR="$(cd "$(dirname "$0")" && pwd)"
PROJECT_DIR="$(dirname "$SCRIPT_DIR")"
cd "$PROJECT_DIR"

# Colors for output
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
RED='\033[0;31m'
NC='\033[0m' # No Color

log_info() { echo -e "${GREEN}[INFO]${NC} $1"; }
log_warn() { echo -e "${YELLOW}[WARN]${NC} $1"; }
log_error() { echo -e "${RED}[ERROR]${NC} $1"; }

# 1. Check prerequisites
log_info "Checking prerequisites..."
command -v python3 >/dev/null 2>&1 || { log_error "python3 is required"; exit 1; }
command -v llama-server >/dev/null 2>&1 || { log_warn "llama-server not found in PATH"; }

# 2. Activate virtual environment
if [ -d ".venv" ]; then
    source .venv/bin/activate
elif [ -d "venv" ]; then
    source venv/bin/activate
else
    log_warn "No virtual environment found, using system Python"
fi

# 3. Verify dependencies
log_info "Verifying dependencies..."
pip check 2>/dev/null || log_warn "Dependencies may have conflicts"

# 4. Check config files exist
for cfg in configs/storage.yaml configs/middleware.yaml configs/watchlist.yaml; do
    if [ ! -f "$cfg" ]; then
        log_warn "Missing config: $cfg"
    fi
done

# 5. Start llama-server (if not already running)
if ! curl -s http://127.0.0.1:8087/health >/dev/null 2>&1; then
    log_info "Starting llama-server..."
    MODEL_PATH="${TRACEALCHEMY_MODEL:-./models/TraceAlchemy-Gemma-4-E4B-Finance-IT.gguf}"
    if [ -f "$MODEL_PATH" ]; then
        llama-server -m "$MODEL_PATH" \
            --port 8087 \
            --ctx-size 32768 \
            --n-gpu-layers 99 \
            --log-format json \
            > logs/llama-server.log 2>&1 &
        LLAMA_PID=$!
        log_info "llama-server started (PID: $LLAMA_PID)"
        # Wait for model to load
        sleep 5
    else
        log_warn "Model not found at $MODEL_PATH — skipping llama-server"
    fi
else
    log_info "llama-server already running"
fi

# 6. Start FastAPI middleware
log_info "Starting FastAPI middleware..."
uvicorn src.middleware.app:app \
    --host 0.0.0.0 \
    --port 8000 \
    --log-level info \
    > logs/middleware.log 2>&1 &
MIDDLEWARE_PID=$!
log_info "Middleware started (PID: $MIDDLEWARE_PID)"

# 7. Verify stack is up
sleep 2
log_info "Verifying stack..."
curl -s http://127.0.0.1:8000/health >/dev/null 2>&1 && \
    log_info "✅ Middleware health check passed" || \
    log_warn "Middleware health check failed"

echo ""
log_info "Stack is running!"
echo "  Middleware:  http://127.0.0.1:8000"
echo "  Docs:        http://127.0.0.1:8000/docs"
echo "  Model:       http://127.0.0.1:8087"
echo ""
echo "To stop: kill $LLAMA_PID $MIDDLEWARE_PID"
```

```python
@app.get("/health", response_model=HealthResponse)
async def health():
    """Enhanced health check with per-source status."""
    if not store:
        raise HTTPException(status_code=503, detail="Store not initialized")

    storage_health = store.heartbeat()
    model_ok = await _check_model_health()

    # Add scheduler status
    from src.scheduler import UnifiedScheduler
    sched = UnifiedScheduler(store=store)
    scheduler_status = sched.status_report()

    # Add per-source freshness summary
    freshness_summary = {}
    for ticker in ["NVDA", "AMD", "AAPL", "MSFT", "META", "CRWD"]:
        try:
            report = store.get_freshness_report(ticker)
            freshness_summary[ticker] = report["overall"]
        except Exception:
            freshness_summary[ticker] = "error"

    return HealthResponse(
        status="ok" if storage_health.get("sqlite") else "degraded",
        storage=storage_health,
        model_available=model_ok,
        scheduler=scheduler_status,
        freshness=freshness_summary,
        version="1.0.0",
    )
```

## Step 4: Add Graceful Degradation for Missing Model

When `llama-server` is not running, the query endpoint should still return useful information — just without model inference:

```python
@app.post("/query", response_model=QueryResponse)
async def query(request: QueryRequest):
    """Query with graceful degradation when model is unavailable."""
    # ... existing intent parsing and retrieval ...

    # Check model availability before calling
    model_available = await _check_model_health()

    if model_available:
        answer_text, citations = await _call_model(...)
    else:
        # Degraded mode: return retrieved data as structured answer
        answer_text = _format_degraded_answer(retrieval, intent)
        citations = []

    return QueryResponse(
        answer=answer_text,
        citations=citations,
        model_available=model_available,
        # ... rest of response ...
    )

def _format_degraded_answer(retrieval: dict, intent: dict) -> str:
    """Format retrieved data as a readable answer when model is unavailable."""
    parts = ["⚠️ Model unavailable — showing raw retrieved data:\n"]
    facts = retrieval.get("facts", [])
    docs = retrieval.get("documents", [])

    if facts:
        parts.append("**Structured Facts:**")
        for f in facts[:5]:
            parts.append(f"- {f.get('metric')}: {f.get('value')} ({f.get('period', 'N/A')})")
        parts.append("")

    if docs:
        parts.append("**Relevant Documents:**")
        for d in docs[:3]:
            parts.append(f"- {d.get('id', 'unknown')} ({d.get('source', 'unknown')})")
        parts.append("")

    parts.append("Start llama-server to get AI-grounded answers.")
    return "\n".join(parts)
```

```python
#!/usr/bin/env python3
"""Validate the RAG system setup — configs, dependencies, endpoints."""

import sys
from pathlib import Path

    if not condition:
        if is_error:
            errors.append(msg)
        else:
            warnings.append(msg)

# 1. Check project structure

for cfg in ["storage.yaml", "middleware.yaml", "watchlist.yaml", "fred.yaml", "gdelt.yaml", "ir.yaml"]:

    import chromadb

check(health.get("sqlite"), "SQLite health check failed", is_error=True)
```

```python
def setup_file_logging(log_dir: str = "logs", level=logging.INFO):
    """Configure file logging with rotation for the RAG system."""
    import os
    from logging.handlers import RotatingFileHandler

    os.makedirs(log_dir, exist_ok=True)

    formatter = logging.Formatter(
        "%(asctime)s [%(levelname)s] %(name)s: %(message)s",
        datefmt="%Y-%m-%d %H:%M:%S",
    )

    # Main log
    handler = RotatingFileHandler(
        os.path.join(log_dir, "rag-system.log"),
        maxBytes=10 * 1024 * 1024,  # 10 MB
        backupCount=5,
    )
    handler.setFormatter(formatter)
    handler.setLevel(level)

    # Error log
    error_handler = RotatingFileHandler(
        os.path.join(log_dir, "rag-errors.log"),
        maxBytes=10 * 1024 * 1024,
        backupCount=3,
    )
    error_handler.setFormatter(formatter)
    error_handler.setLevel(logging.WARNING)

    root = logging.getLogger()
    root.addHandler(handler)
    root.addHandler(error_handler)
    root.setLevel(level)

    return handler, error_handler
```
